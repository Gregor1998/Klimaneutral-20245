def calculateConsumption(consumption_development_per_year, lastprofile_dict, directory_heatpump_consumption):
    """
    Calculate energy consumption with heat pumps and EV profiles subtracted but not re-added.
    
    Args:
        consumption_development_per_year: Dictionary with yearly growth factors
        lastprofile_dict: Dictionary containing load profiles for EVs by sector and day type
        directory_heatpump_consumption: Dictionary with heat pump consumption data by year
        
    Returns:
        Dictionary containing consumption dataframes for each year
    """
    consumption_year = config.params.consumption_year  # Base year for calculations
    directory_yearly_consumption = getData("Consumption", consumption_year)  # Load consumption data
    addTimeInformation(directory_yearly_consumption[consumption_year])  # Add columns for date/time info
    base_heatpump_lp = directory_heatpump_consumption.get(consumption_year + 1)  # Get heat pump load profile

    # Helper function to apply lastprofile values (subtract or add)
    def apply_lastprofile(df, lastprofile, heatpump_profile, mode="subtract"):
        """
        Apply EV and heat pump load profiles to consumption data.
        
        Args:
            df: DataFrame with consumption data
            lastprofile: Dictionary with EV load profiles
            heatpump_profile: Heat pump load profile data
            mode: "subtract" to remove profiles, "add" to include them
            
        Returns:
            DataFrame with adjusted consumption values
        """
        # Define mapping of weekdays to profiles
        profile_mapping = {
            "6": "saturday",
            "7": "sunday",
            "1": "workday",
            "2": "workday",
            "3": "workday",
            "4": "workday",
            "5": "workday"
        }

        # Map weekdays to profile names
        df['profile'] = df['Weekday'].map(profile_mapping)

        # Drop rows with weekdays not in the mapping
        df = df[df['profile'].notna()]

        # Function to compute the adjustment value for each row
        def compute_adjustment(row):
            profile_name = row['profile']
            lastprofil_idx = row.name % len(lastprofile['Wohnen'][profile_name])
            
            # Get EV consumption from different sectors
            lp_wohnen = lastprofile['Wohnen'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']
            lp_buro = lastprofile['Büro'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']
            lp_public = lastprofile['Öffentliche_Ladepunkte'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']

            # Calculate total EV and heat pump consumption (converted to MWh)
            lp_eautos_sum = lp_wohnen + lp_buro + lp_public
            adjustment_value = (lp_eautos_sum / 1000) + heatpump_profile.loc[row.name, 'Verbrauch in MWh']

            # Apply adjustment based on mode
            if mode == "subtract":
                return row['Gesamtverbrauch'] - adjustment_value
            elif mode == "add":
                return row['Gesamtverbrauch'] + adjustment_value
            else:
                return row['Gesamtverbrauch']  # No adjustment if mode is invalid

        # Apply the adjustment computation function
        df['Gesamtverbrauch'] = df.apply(compute_adjustment, axis=1)

        # Drop the temporary profile column
        df.drop(columns=['profile'], inplace=True)

        return df

    # STEP 1: Subtract EV and heat pump profiles from base year consumption
    directory_yearly_consumption[consumption_year] = apply_lastprofile(
        directory_yearly_consumption[consumption_year],
        lastprofile_dict[consumption_year],
        base_heatpump_lp,
        mode="subtract"
    )

    # Save intermediate result for validation
    directory_yearly_consumption[consumption_year].to_csv("CSV/testing/consumption_2023.csv")

    # STEP 2: Extrapolate consumption for all forecast years
    for year in range(config.params.start_year_simulation, config.params.end_year_simulation + 1):
        prev_year_df = directory_yearly_consumption.get(year - 1).copy()

        # Project consumption for this year based on previous year
        extrapolated_data = Extrapolation_Consumption(
            prev_year_df, year, None, None, None, consumption_development_per_year.get(year)
        )
        
        # Store the extrapolated data
        directory_yearly_consumption[extrapolated_data.year] = extrapolated_data.df

    return directory_yearly_consumption

def calculateConsumption_lastprofile(consumption_development_per_year, lastprofile_dict, directory_heatpump_consumption):
    """
    Complete consumption calculation: subtract profiles, extrapolate, then add profiles back.
    
    This function follows the same initial steps as calculateConsumption() but adds
    a crucial third step to re-add the EV and heat pump profiles for each year.
    
    Args:
        consumption_development_per_year: Dictionary with yearly growth factors
        lastprofile_dict: Dictionary containing load profiles for EVs by sector and day type
        directory_heatpump_consumption: Dictionary with heat pump consumption data by year
        
    Returns:
        Dictionary containing consumption dataframes for each year with profiles applied
    """
    # Steps 1-2 are identical to the previous function
    consumption_year = config.params.consumption_year
    directory_yearly_consumption = getData("Consumption", consumption_year)
    addTimeInformation(directory_yearly_consumption[consumption_year])
    base_heatpump_lp = directory_heatpump_consumption.get(consumption_year + 1)

    # Helper function definition (identical to the one in calculateConsumption)
    def apply_lastprofile(df, lastprofile, heatpump_profile, mode="subtract"):
        # Same implementation as above - helper function for applying/removing profiles
        profile_mapping = {
            "6": "saturday",
            "7": "sunday",
            "1": "workday",
            "2": "workday",
            "3": "workday",
            "4": "workday",
            "5": "workday"
        }

        df['profile'] = df['Weekday'].map(profile_mapping)
        df = df[df['profile'].notna()]

        def compute_adjustment(row):
            profile_name = row['profile']
            lastprofil_idx = row.name % len(lastprofile['Wohnen'][profile_name])
            
            lp_wohnen = lastprofile['Wohnen'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']
            lp_buro = lastprofile['Büro'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']
            lp_public = lastprofile['Öffentliche_Ladepunkte'][profile_name].loc[lastprofil_idx, 'Strombedarf (kWh)']

            lp_eautos_sum = lp_wohnen + lp_buro + lp_public
            adjustment_value = (lp_eautos_sum / 1000) + heatpump_profile.loc[row.name, 'Verbrauch in MWh']

            if mode == "subtract":
                return row['Gesamtverbrauch'] - adjustment_value
            elif mode == "add":
                return row['Gesamtverbrauch'] + adjustment_value
            else:
                return row['Gesamtverbrauch']

        df['Gesamtverbrauch'] = df.apply(compute_adjustment, axis=1)
        df.drop(columns=['profile'], inplace=True)
        return df

    # STEP 1: Subtract profiles from base year (same as previous function)
    directory_yearly_consumption[consumption_year] = apply_lastprofile(
        directory_yearly_consumption[consumption_year],
        lastprofile_dict[consumption_year],
        base_heatpump_lp,
        mode="subtract"
    )

    directory_yearly_consumption[consumption_year].to_csv("CSV/testing/consumption_2023.csv")

    # STEP 2: Extrapolate for all years (same as previous function)
    for year in range(config.params.start_year_simulation, config.params.end_year_simulation + 1):
        prev_year_df = directory_yearly_consumption.get(year - 1).copy()

        extrapolated_data = Extrapolation_Consumption(
            prev_year_df, year, None, None, None, consumption_development_per_year.get(year)
        )

        directory_yearly_consumption[extrapolated_data.year] = extrapolated_data.df

        directory_yearly_consumption[extrapolated_data.year].to_csv(f"CSV/testing/consumption_after_hochrechnung_{year}.csv")

    # STEP 3: Add back profiles for each year (unique to this function)
    for year in range(config.params.start_year_simulation, config.params.end_year_simulation + 1):
        directory_yearly_consumption[year] = apply_lastprofile(
            directory_yearly_consumption[year],
            lastprofile_dict[year],
            directory_heatpump_consumption.get(year, base_heatpump_lp),  # Use year's profile or base profile
            mode="add"  # Add profiles back to get total consumption
        )
        directory_yearly_consumption[year].to_csv(f"CSV/testing/consumption_final_mit_lp_{year}.csv")

    return directory_yearly_consumption

def load_profile_heatpump(installed_heatpumps, expected_heatpumps, first_year, end_year):
    """
    Calculates the electricity load profile for heat pumps over multiple years.
    
    Parameters:
    - installed_heatpumps: Initial number of installed heat pumps at the start year
    - expected_heatpumps: Target number of heat pumps by the end of the period
    - first_year: Starting year for the calculation
    - end_year: Ending year for the calculation
    
    Returns:
    - Dictionary containing yearly load profiles as DataFrames
    """
    # Calculate the time span in years
    difference_year = end_year - first_year
    
    # Calculate total heat pump growth over the period
    difference_heatpumps = expected_heatpumps - installed_heatpumps
    
    # Track the current number of installed heat pumps
    current_installed = installed_heatpumps

    # Dictionary to store yearly load profiles
    directory_heatpump_comsumption = {}

    # Load the reference heat pump load profile from data source
    df = getData("Heatpump")
    
    # Calculate daily growth rate of heat pump installations
    daily_expansion_rate_heatpump = difference_heatpumps / (difference_year*365)

    # Process each year in the specified range
    for year in range(first_year, end_year+1):
        # Initialize DataFrame for the year with 35040 entries (96 quarter-hours × 365 days)
        heatpump_df = pd.DataFrame({"Verbrauch in MWh": [0.0]*35040})
        
        # Add timestamp information to the DataFrame
        heatpump_df = addTimePerformance(heatpump_df, year)
                                
        # Calculate load for each day of the year
        for day in range(365):
            # Calculate indices for the current day
            start_index = day*96  # 96 quarter-hours per day
            end_index = start_index + 96

            # Update the number of installed heat pumps
            current_installed += daily_expansion_rate_heatpump

            # Calculate consumption in MWh based on the reference profile and number of heat pumps
            heatpump_df.loc[start_index:end_index, "Verbrauch in MWh"] = df["Lastprofil"][start_index:end_index] * current_installed/1000

        # Store the annual load profile in the dictionary
        directory_heatpump_comsumption[year] = heatpump_df
        
        # Export the annual load profile to a CSV file
        directory_heatpump_comsumption[year].to_csv(f"CSV/Lastprofile/waermepumpe/Hochrechnung/Heatpump_{year}.csv", index=False)

    # Return the dictionary containing all annual load profiles
    return directory_heatpump_comsumption
